ORTHO_GRIDMAP ubuntu commands

----------------------------------------------------------------------------------------------------------------------------
(GRIDMAP)

cd ~/ros2_ws
colcon build
source install/setup.bash

//need path!
ros2 run ortho_gridmap ortho_gridmap_node \
  --ros-args \
  -p tif_path:="/mnt/d/   ...     " \
  -p grid_resolution:=0.3

----------------------------------------------------------------------------------------------------------------------------
(GUI)

cd ~/ros2_ws
colcon build
source install/setup.bash

ros2 run gridmap_editor map_editor

//mouse interactions:
  - left button -> FREE
  - middle button (or shift + left button) -> UNKNOWN
  - right button -> OCCUPIED

----------------------------------------------------------------------------------------------------------------------------

source ~/ros2_ws/install/setup.bash
ros2 run foxglove_bridge foxglove_bridge
